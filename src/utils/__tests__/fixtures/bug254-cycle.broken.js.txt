  function resolveHostWindow() {
    let current = window;
    try {
      for (let i = 0; i < 6; i++) {
        if (!current.parent || current.parent === current) break;
        try {
          if (current.parent.document && current.parent.document.body) current = current.parent;
          else break;
        } catch (secErr) { break; }
      }
    } catch (_) {}
    return current;
  }

  function hashName(key) {
    const s = String(key || '');
    let h = 2166136261;
    for (let i = 0; i < s.length; i++) h = Math.imul(h ^ s.charCodeAt(i), 16777619);
    return h >>> 0;
  }

  function raceNameOf(stat, name) {
    const arch = stat && stat['Hồ Sơ Nhân Vật Nữ'] && stat['Hồ Sơ Nhân Vật Nữ'][name];

  // Ngày解析：返回天数序号、年份与精度。无年份时继承参考年份，月日超出范围时判为无效
  function buildDate(year, month, day, precision) {
    if (!Number.isFinite(year) || !Number.isFinite(month) || !Number.isFinite(day)) return null;
    if (month < 1 || month > 12 || day < 1 || day > 31) return null;
    const t = Date.UTC(year, month - 1, day);
    if (!Number.isFinite(t)) return null;
    const d = new Date(t);
    if (d.getUTCFullYear() !== year || d.getUTCMonth() + 1 !== month || d.getUTCDate() !== day) return null;
    return { idx: Math.floor(t / 86400000), year: year, precision: precision };
  }

  function parseDateValue(str, refYear) {
    const s = asStr(str);
    if (!s || s === 'Không Rõ' || s === '无' || s === 'Chưa Từng' || s === '--') return null;
    let m = s.match(/(\d{4})\s*年\s*(\d{1,2})\s*月\s*(\d{1,2})\s*[日号]/);
    if (m) return buildDate(+m[1], +m[2], +m[3], 'day');
    m = s.match(/(\d{4})\s*年\s*(\d{1,2})\s*月/);
    if (m) return buildDate(+m[1], +m[2], 1, 'month');
    m = s.match(/(\d{1,2})\s*月\s*(\d{1,2})\s*[日号]/);
    if (m && Number.isFinite(refYear)) return buildDate(refYear, +m[1], +m[2], 'day');
    m = s.match(/(\d{4})\s*[-/.]\s*(\d{1,2})\s*[-/.]\s*(\d{1,2})/);
    if (m) return buildDate(+m[1], +m[2], +m[3], 'day');
    m = s.match(/(?:^|\D)(\d{4})(\d{2})(\d{2})(?:\D|$)/);
    if (m) return buildDate(+m[1], +m[2], +m[3], 'day');
    return null;
  }

  function formatDateIdx(days) {
    const d = new Date(days * 86400000);
    const p2 = n => String(n).padStart(2, '0');
    return d.getUTCFullYear() + '年' + p2(d.getUTCMonth() + 1) + '月' + p2(d.getUTCDate()) + '日';
  }

  // Thời Gian基准解析：按四层Nguồn Gốc给出今天的天数序号，回传Nguồn Gốc标记供日志使用
  function resolveTodayIdx(stat, sys) {
    const anchor = parseDateValue(sys && sys['Ngày Xử Lý Lần Trước'], null);
    const refYear = anchor ? anchor.year : new Date().getUTCFullYear();
    const today = parseDateValue(stat && stat['Ngày'], refYear);
    if (today) return { idx: today.idx, source: today.precision === 'month' ? 'Ngày月精度' : 'Ngày' };
    if (anchor) return { idx: anchor.idx, source: 'Dùng Lại Điểm Neo' };
    return { idx: Math.floor(Date.now() / 86400000), source: 'Ngày Hiện Tại Hệ Thống' };
  }

  function derivedParams(name, race) {
    const h = hashName('cycle#' + name);
    const base = RACE_CYCLE_BASE[race];
    if (base) {
      if (base[0] <= 0) return { L: 0, P: 0 };
      return { L: Math.max(1, base[0] + (h % 5) - 2), P: Math.max(0, base[1] + (Math.floor(h / 15) % 3) - 1) };
    }
    const Lv = 21 + (h % 15);
    const Pv = 3 + (Math.floor(h / 15) % 5);
    return { L: Lv, P: Pv };
  }

  function derivedGestationDays(name, race) {
    const h = hashName('gestation#' + name);
    const base = RACE_CYCLE_BASE[race];
    if (base) return base[2] > 0 ? Math.max(30, base[2] + (h % 11) - 5) : 0;
    return 259 + (h % 17);
  }

  function layoutOf(rec) {
    const Lv = clamp(Math.round(asNumber(rec['Độ Dài Chu Kỳ'], 28)), 0, 400);
    const Pv = clamp(Math.round(asNumber(rec['Độ Dài Kỳ Kinh'], 5)), 0, 20);
    const Ov = Lv > 0 ? clamp(Lv - 14, Pv + 3, Lv - 2) : 0;
    return { L: Lv, P: Pv, O: Ov };
  }

  function computePhase(day, lay) {
    if (day <= lay.P) return PHASE_MENSTRUATION;
    if (day < lay.O - 2) return PHASE_SAFE;
    if (day <= lay.O + 1) return PHASE_OVULATION;
    return PHASE_LUTEAL;
  }

  function nextPhaseInfo(day, lay) {
    if (day <= lay.P) return { remain: Math.max(0, lay.P + 1 - day), next: PHASE_SAFE };
    if (day < lay.O - 2) return { remain: Math.max(0, (lay.O - 2) - day), next: PHASE_OVULATION };
    if (day <= lay.O + 1) return { remain: Math.max(0, (lay.O + 2) - day), next: PHASE_LUTEAL };
    return { remain: Math.max(0, lay.L + 1 - day), next: PHASE_MENSTRUATION };
  }

  function refreshDerived(rec) {
    const lay = layoutOf(rec);
    rec['Độ Dài Chu Kỳ'] = lay.L;
    rec['Độ Dài Kỳ Kinh'] = lay.P;
    rec['Ngày Rụng Trứng'] = lay.O;
    if (lay.L <= 0) {
      rec['Ngày Thứ Mấy Chu Kỳ'] = 0;
      rec['Giai Đoạn Hiện Tại'] = PHASE_SAFE;
      rec['Giai Đoạn Tiếp Theo'] = PHASE_SAFE;
      rec['Số Ngày Đến Giai Đoạn Tiếp Theo'] = 0;
      return;
    }
    rec['Ngày Thứ Mấy Chu Kỳ'] = clamp(Math.round(asNumber(rec['Ngày Thứ Mấy Chu Kỳ'], 1)) || 1, 1, lay.L);
    rec['Giai Đoạn Hiện Tại'] = computePhase(rec['Ngày Thứ Mấy Chu Kỳ'], lay);
    const ni = nextPhaseInfo(rec['Ngày Thứ Mấy Chu Kỳ'], lay);
    rec['Giai Đoạn Tiếp Theo'] = ni.next;
    rec['Số Ngày Đến Giai Đoạn Tiếp Theo'] = ni.remain;
  }

  function representativeDay(phase, lay) {
    if (phase === PHASE_MENSTRUATION) return clamp(2, 1, lay.P);
    if (phase === PHASE_OVULATION) return lay.O;
    if (phase === PHASE_LUTEAL) return clamp(lay.O + 3, lay.O + 2, lay.L);
    if (phase === PHASE_SAFE) {
      const lo = lay.P + 1;
      const hi = Math.max(lo, lay.O - 3);
      return clamp(lay.P + 2, lo, hi);
    }
    return -1;
  }

  function femaleCopies(stat, name) {
    const out = [];
    ['Nữ Tương Tác Tại Chỗ', 'Hồ Sơ Nhân Vật Nữ'].forEach(k => {
      const copy = isPlainObject(stat[k]) ? stat[k][name] : undefined;
      if (isPlainObject(copy)) out.push(copy);
    });
    return out;
  }

  function syncPrivate(stat, name, patch) {
    femaleCopies(stat, name).forEach(copy => {
      if (!isPlainObject(copy['Hồ Sơ Riêng Tư'])) return;
      Object.keys(patch).forEach(k => { copy['Hồ Sơ Riêng Tư'][k] = patch[k]; });
    });
  }

  // Ngày Dự Sinh只从有证据的Nguồn Gốc推算，证据Cấp Độ写入 source 供Trạng Thái栏区分展示
  function findDueDate(name, stat, rec) {
    for (const copy of femaleCopies(stat, name)) {
      if (isPlainObject(copy['Hồ Sơ Riêng Tư']) && asStr(copy['Hồ Sơ Riêng Tư']['Loại Ngày Dự Sinh']) === 'Xác Chẩn') {
        const d = parseDateValue(copy['Hồ Sơ Riêng Tư']['Ngày Dự Sinh'], null);
        if (d !== null) return { idx: d.idx, type: 'Xác Chẩn', source: 'Dài Hạn档案Xác Chẩn' };
      }
    }
    const events = isPlainObject(stat?.['Sự Kiện Theo Dõi Nhân Vật']) ? Object.values(stat['Sự Kiện Theo Dõi Nhân Vật']) : [];
    for (const ev of events) {
      if (!isPlainObject(ev)) continue;
      if (asStr(ev['Nhân Vật']) !== name && asStr(ev['Đối Tượng Liên Quan']) !== name) continue;
      if (asStr(ev['Loại Ngày Dự Sinh']) === 'Xác Chẩn' || asStr(ev['Trạng Thái Hiện Tại']) === 'Xác Chẩn') {
        const d = parseDateValue(ev['Ngày Dự Sinh'], null);
        if (d !== null) return { idx: d.idx, type: 'Xác Chẩn', source: '追踪Sự KiệnXác Chẩn' };
        const note = parseDateValue(ev['Ghi Chú Xử Lý'], null);
        if (note !== null) return { idx: note.idx, type: 'Xác Chẩn', source: '追踪Sự KiệnGhi Chú' };
      }
    }
    if (rec && asStr(rec['Trạng Thái Ngày Dự Sinh']) === 'Xác Chẩn') {
      const d = parseDateValue(rec['Ngày Dự Sinh'], null);
      if (d !== null) return { idx: d.idx, type: 'Xác Chẩn', source: '周期记录Xác Chẩn' };
    }
    for (const copy of femaleCopies(stat, name)) {
      if (isPlainObject(copy['Hồ Sơ Riêng Tư']) && asStr(copy['Hồ Sơ Riêng Tư']['Loại Ngày Dự Sinh']) === 'Dự Đoán') {
        const d = parseDateValue(copy['Hồ Sơ Riêng Tư']['Ngày Dự Sinh'], null);
        if (d !== null) return { idx: d.idx, type: 'Dự Đoán', source: 'Dài Hạn档案Dự Đoán' };
      }
    }
    if (rec && asStr(rec['Trạng Thái Ngày Dự Sinh']) === 'Dự Đoán') {
      const d = parseDateValue(rec['Ngày Dự Sinh'], null);
      if (d !== null) return { idx: d.idx, type: 'Dự Đoán', source: '周期记录Dự Đoán' };
    }
    const gDays = derivedGestationDays(name, raceNameOf(stat, name));
    for (const copy of femaleCopies(stat, name)) {
      if (isPlainObject(copy['Hồ Sơ Riêng Tư'])) {
        const sexDate = parseDateValue(copy['Hồ Sơ Riêng Tư']['Thời Gian Xuất Tinh Trong Lần Trước'], null);
        if (sexDate !== null) return { idx: sexDate.idx + gDays, type: 'Dự Đoán', source: '内射Thời Gian推算' };
      }
    }
    return null;
  }

  function initRecord(name, sys, todayIdx, stat) {
    const dp = derivedParams(name, raceNameOf(stat, name));
    const rec = {
      'Độ Dài Chu Kỳ': dp.L,
      'Độ Dài Kỳ Kinh': dp.P,
      'Ngày Rụng Trứng': clamp(dp.L - 14, dp.P + 3, dp.L - 2),
      'Ngày Thứ Mấy Chu Kỳ': 1,
      'Giai Đoạn Hiện Tại': PHASE_SAFE,
      'Giai Đoạn Tiếp Theo': PHASE_OVULATION,
      'Số Ngày Đến Giai Đoạn Tiếp Theo': 0,
      'Ngày Bắt Đầu Chu Kỳ Này': '',
      'Đang Mang Thai': false,
      'Ngày Dự Sinh': '',
      'Trạng Thái Ngày Dự Sinh': 'Chưa Mang Thai',
      'Chu Kỳ Lịch Sử': {}
    };
    let startDay = -1;
    femaleCopies(stat, name).forEach(copy => {
      if (startDay > 0 || !isPlainObject(copy['Hồ Sơ Riêng Tư'])) return;
      const guess = representativeDay(asStr(copy['Hồ Sơ Riêng Tư']['Kỳ Sinh Lý']), layoutOf(rec));
      if (guess > 0) startDay = guess;
    });
    if (startDay < 0 && layoutOf(rec).L > 0) startDay = 1 + (hashName('phase#' + name) % layoutOf(rec).L);
    if (startDay < 0) startDay = 1;
    rec['Ngày Thứ Mấy Chu Kỳ'] = startDay;
    refreshDerived(rec);
    rec['Ngày Bắt Đầu Chu Kỳ Này'] = formatDateIdx(todayIdx - (rec['Ngày Thứ Mấy Chu Kỳ'] - 1));
    sys['Nhân Vật'][name] = rec;
    syncPrivate(stat, name, { 'Kỳ Sinh Lý': rec['Giai Đoạn Hiện Tại'], 'Ngày Đặc Trưng Rụng Trứng': rec['Ngày Rụng Trứng'] });
    return rec;
  }

  function pushHistory(rec) {
    if (!isPlainObject(rec['Chu Kỳ Lịch Sử'])) rec['Chu Kỳ Lịch Sử'] = {};
    const key = asStr(rec['Ngày Bắt Đầu Chu Kỳ Này']);
    if (!key) return;
    rec['Chu Kỳ Lịch Sử'][key] = { 'Độ Dài': clamp(Math.round(asNumber(rec['Độ Dài Chu Kỳ'], 28)), 0, 400) };
    const keys = Object.keys(rec['Chu Kỳ Lịch Sử']);
    while (keys.length > 8) {
      delete rec['Chu Kỳ Lịch Sử'][keys[0]];
      keys.shift();
    }
  }

  function advanceRecord(rec, days, todayIdx) {
    const lay = layoutOf(rec);
    if (lay.L <= 0) { refreshDerived(rec); return; }
    let day = Math.round(asNumber(rec['Ngày Thứ Mấy Chu Kỳ'], 1)) + days;
    const cycleCount = Math.ceil(days / lay.L) + 2;
    let guard = 0;
    while (day > lay.L && guard < cycleCount) {
      day -= lay.L;
      guard++;
      pushHistory(rec);
      const base = parseDateValue(rec['Ngày Bắt Đầu Chu Kỳ Này'], null);
      rec['Ngày Bắt Đầu Chu Kỳ Này'] = formatDateIdx((base === null ? todayIdx : base.idx) + lay.L);
    }
    if (day > lay.L) day = ((day - 1) % lay.L) + 1;
    rec['Ngày Thứ Mấy Chu Kỳ'] = clamp(day, 1, lay.L);
    refreshDerived(rec);
  }

  function restartFromMenstruation(rec, todayIdx) {
    rec['Đang Mang Thai'] = false;
    rec['Ngày Dự Sinh'] = '';
    rec['Trạng Thái Ngày Dự Sinh'] = 'Chưa Mang Thai';
    rec['Ngày Thứ Mấy Chu Kỳ'] = 1;
    rec['Ngày Bắt Đầu Chu Kỳ Này'] = formatDateIdx(todayIdx);
    refreshDerived(rec);
  }

  function copyIsPregnant(copy) {
    const p = copy && copy['Hồ Sơ Riêng Tư'];
    if (!isPlainObject(p)) return false;
    if (asStr(p['Có Thai Không']) === '是') return true;
    if (asStr(p['Kỳ Sinh Lý']) === PHASE_PREGNANT && asStr(p['Có Thai Không']) !== '否') return true;
    return false;
  }

  const PREGNANT_STATES = ['Đã Xác Nhận Mang Thai', 'Tiến Triển Thai Kỳ', 'Chờ Xử Lý Sắp Sinh'];

  // 追踪Sự Kiện作为妊娠Trạng Thái的第二个Nguồn Gốc，Hồ Sơ Riêng Tư漏写时依然成立
  function trackedPregnancy(stat, name) {
    const events = isPlainObject(stat && stat['Sự Kiện Theo Dõi Nhân Vật']) ? stat['Sự Kiện Theo Dõi Nhân Vật'] : null;
    if (!events) return false;
    return Object.values(events).some(ev => {
      if (!isPlainObject(ev)) return false;
      if (asStr(ev['Nhân Vật']) !== name && asStr(ev['Đối Tượng Liên Quan']) !== name) return false;
      return PREGNANT_STATES.indexOf(asStr(ev['Trạng Thái Hiện Tại'])) >= 0;
    });
  }

  // 汇总已到期的追踪Sự Kiện，供正文优先处理
  function collectDueChecks(stat, todayIdx) {
    const out = [];
    const events = isPlainObject(stat && stat['Sự Kiện Theo Dõi Nhân Vật']) ? stat['Sự Kiện Theo Dõi Nhân Vật'] : null;
    if (!events) return out;
    Object.keys(events).forEach(key => {
      const ev = events[key];
      if (!isPlainObject(ev)) return;
      if (PREGNANT_STATES.indexOf(asStr(ev['Trạng Thái Hiện Tại'])) >= 0) return;
      const check = parseDateValue(ev['Ngày Kiểm Tra Lần Tới'], null);
      const due = parseDateValue(ev['Ngày Hạn Chót Xác Nhận'], null);
      const reached = (check !== null && todayIdx >= check.idx) || (due !== null && todayIdx >= due.idx);
      if (reached) out.push({ 'Sự Kiện': key, 'Nhân Vật': asStr(ev['Nhân Vật']), 'Tỷ Lệ Mang Thai': asStr(ev['Tỷ Lệ Mang Thai']) });
    });
    return out;
  }

  // 汇总Đã Xác Nhận Mang Thai但尚未告知主角的女性，供正文安排通知
  function collectPendingNotice(stat) {
    const out = [];
    const names = [];
    const seen = {};
    ['Nữ Tương Tác Tại Chỗ', 'Hồ Sơ Nhân Vật Nữ'].forEach(k => {
      const bucket = stat && stat[k];
      if (!isPlainObject(bucket)) return;
      Object.keys(bucket).forEach(n => { if (n && !seen[n]) { seen[n] = true; names.push(n); } });
    });
    names.forEach(name => {
      const copies = femaleCopies(stat, name);
      const told = copies.some(copy => isPlainObject(copy['Hồ Sơ Riêng Tư']) && asStr(copy['Hồ Sơ Riêng Tư']['Đã Thông Báo Mang Thai']) === '是');
      if (told) return;
      const pregnant = copies.some(copyIsPregnant) || trackedPregnancy(stat, name);
      if (pregnant) out.push(name);
    });
    return out;
  }

  // 把Xác ChẩnNgày Dự Sinh回写到追踪Sự Kiện，并记录证据Nguồn Gốc
  function syncDueTracking(stat, name, dueStr, source) {
    if (!isPlainObject(stat?.['Sự Kiện Theo Dõi Nhân Vật'])) return;
    Object.values(stat['Sự Kiện Theo Dõi Nhân Vật']).forEach(ev => {
      if (!isPlainObject(ev)) return;
      if (asStr(ev['Nhân Vật']) !== name && asStr(ev['Đối Tượng Liên Quan']) !== name) return;
      if (!parseDateValue(ev['Ngày Dự Sinh'], null)) ev['Ngày Dự Sinh'] = dueStr;
      if (source) ev['Nguồn Gốc Ngày Dự Sinh'] = source;
    });
  }

  function collectNames(stat, sources) {
    const out = [];
    const seen = {};
    sources.forEach(k => {
      const bucket = stat[k];
      if (!isPlainObject(bucket)) return;
      Object.keys(bucket).forEach(n => {
        const s = asStr(n);
        if (!s || seen[s] || s === 'Không Rõ') return;
        seen[s] = true;
        out.push(s);
      });
    });
    return out;
  }

  function isUsableName(value) {
    const s = asStr(value);
    if (!s || s.length > 30) return false;
    if (HERO_ALIASES.indexOf(s) >= 0) return false;
    if (/^(?:都市神[豪壕]|kaidan|system|TH-)/i.test(s)) return false;
    return true;
  }

  // 主角名：先读变量，再上溯宿主窗口，取值链与 Bridge 的 resolveHostUserName 一致
  function resolveHeroName(stat) {
    const u = (stat && stat.user) || {};
    const keys = ['姓名', '用户名', '玩家名', 'name'];
    for (let i = 0; i < keys.length; i++) if (isUsableName(u[keys[i]])) return asStr(u[keys[i]]);
    try {
      const host = HOST_WINDOW || {};
      if (host.SillyTavern && typeof host.SillyTavern.getContext === 'function') {
        const ctx = host.SillyTavern.getContext();
        if (ctx && isUsableName(ctx.name2)) return asStr(ctx.name2);
      }
      if (isUsableName(host.user_name)) return asStr(host.user_name);
      if (isUsableName(host.name2)) return asStr(host.name2);
      if (typeof host.substituteParams === 'function') {
        const sub = host.substituteParams('{{user}}');
        if (isUsableName(sub)) return asStr(sub);
      }
    } catch (_) {}
    return '';
  }

  // Theo Dõi Thụ Thai里的男方：Đối Tượng Liên Quan / Nguồn Gốc Người Cha / 父亲 只当「对方」用，绝不据此建档
  function collectPartnerNames(stat) {
    const out = {};
    if (!isPlainObject(stat['Sự Kiện Theo Dõi Nhân Vật'])) return out;
    Object.values(stat['Sự Kiện Theo Dõi Nhân Vật']).forEach(ev => {
      if (!isPlainObject(ev)) return;
      [ev['Đối Tượng Liên Quan'], ev['Nguồn Gốc Người Cha'], ev.父亲].forEach(v => {
        const s = asStr(v);
        if (isUsableName(s)) out[s] = true;
      });
    });
    return out;
  }

  function isFemaleEvidence(name, g) {
    if (!isUsableName(name)) return false;
    if (g.male[name]) return false;
    if (g.hero && name === g.hero) return false;
    if (g.archived[name] || g.trackedFemale[name]) return true;
    return false;
  }

  function buildGenderContext(stat) {
    const hero = resolveHeroName(stat);
    const male = {};
    collectNames(stat, MALE_SOURCES).forEach(n => { male[n] = true; });
    const partner = collectPartnerNames(stat);
    const archived = {};
    collectNames(stat, FEMALE_SOURCES).forEach(n => { archived[n] = true; });
    const trackedFemale = {};
    if (isPlainObject(stat['Sự Kiện Theo Dõi Nhân Vật'])) {
      Object.values(stat['Sự Kiện Theo Dõi Nhân Vật']).forEach(ev => {
        if (!isPlainObject(ev)) return;
        const s = asStr(ev['Nhân Vật']);
        if (isUsableName(s)) trackedFemale[s] = true;
      });
    }
    const g = { hero, male, partner, archived, trackedFemale, roster: [] };
    const seen = {};
    Object.keys(archived).forEach(n => { if (!seen[n]) { seen[n] = true; g.roster.push(n); } });
    Object.keys(trackedFemale).forEach(n => { if (!seen[n]) { seen[n] = true; g.roster.push(n); } });
    g.roster = g.roster.filter(n => isFemaleEvidence(n, g));
    return g;
  }

  // 自愈：把已被证伪为男性 / 主角的记录摘掉，只在有反证时动手
  function pruneInvalidRecords(sys, g) {
    const removed = [];
    Object.keys(sys['Nhân Vật']).forEach(name => {
      if (!isPlainObject(sys['Nhân Vật'][name])) return;
      if (isFemaleEvidence(name, g)) return;
      const maleEvidence = !!g.male[name]
        || (!!g.hero && name === g.hero)
        || (!!g.partner[name] && !g.archived[name] && !g.trackedFemale[name]);
      if (!maleEvidence) return;
      delete sys['Nhân Vật'][name];
      removed.push(name);
    });
    return removed;
  }

  function maintainCycles(stat) {
    if (!isPlainObject(stat)) return '';
    if (!isPlainObject(stat['Hệ Thống Chu Kỳ Sinh Lý'])) stat['Hệ Thống Chu Kỳ Sinh Lý'] = {};
    const sys = stat['Hệ Thống Chu Kỳ Sinh Lý'];
    if (!isPlainObject(sys['Nhân Vật'])) sys['Nhân Vật'] = {};

    const resolved = resolveTodayIdx(stat, sys);
    const todayIdx = resolved.idx;
    const anchor = parseDateValue(sys['Ngày Xử Lý Lần Trước'], null);
    const raw = asStr(stat['Ngày']);

    let D = anchor === null ? 0 : todayIdx - anchor.idx;
    if (D < 0) {
      console.warn('[都市神壕 生理周期调度] Ngày早于锚点，按 0 处理', { Ngày原文: raw, 基准Nguồn Gốc: resolved.source });
      D = 0;
    }
    if (D > 400) {
      console.warn('[都市神壕 生理周期调度] 单轮跨度超过上限，收束到 400 天', { Ngày原文: raw, 原始差值: D });
      D = 400;
    } else if (D > 60) {
      console.warn('[都市神壕 生理周期调度] 单轮跨度较大', { Ngày原文: raw, 天数: D });
    }

    if (resolved.source === 'Dùng Lại Điểm Neo' || resolved.source === 'Ngày Hiện Tại Hệ Thống') {
      console.warn('[都市神壕 生理周期调度] Ngày未能解析，本轮不推进', { Ngày原文: raw, 基准Nguồn Gốc: resolved.source });
    }
    sys['Ngày Xử Lý Lần Trước'] = formatDateIdx(todayIdx);

    const gender = buildGenderContext(stat);
    const names = gender.roster;
    const pruned = pruneInvalidRecords(sys, gender);

    const changed = [];
    if (pruned.length) changed.push('清理误收录(' + pruned.join('/') + ')');
    names.forEach(name => {
      let rec = isPlainObject(sys['Nhân Vật'][name]) ? sys['Nhân Vật'][name] : null;
      const pregnant = femaleCopies(stat, name).some(copyIsPregnant) || trackedPregnancy(stat, name);

      if (pregnant) {
        if (!rec) rec = initRecord(name, sys, todayIdx, stat);
        rec['Đang Mang Thai'] = true;
        rec['Giai Đoạn Hiện Tại'] = PHASE_PREGNANT;
        rec['Giai Đoạn Tiếp Theo'] = PHASE_MENSTRUATION;
        rec['Số Ngày Đến Giai Đoạn Tiếp Theo'] = 0;
        const dueInfo = findDueDate(name, stat, rec);
        if (dueInfo !== null) {
          const dueStr = formatDateIdx(dueInfo.idx);
          rec['Ngày Dự Sinh'] = dueStr;
          rec['Trạng Thái Ngày Dự Sinh'] = dueInfo.type;
          syncPrivate(stat, name, { 'Kỳ Sinh Lý': PHASE_PREGNANT, 'Có Thai Không': '是', 'Ngày Dự Sinh': dueStr, 'Loại Ngày Dự Sinh': dueInfo.type });
          syncDueTracking(stat, name, dueStr, dueInfo.source);
          const diff = dueInfo.idx - todayIdx;
          const tail = diff > 0 ? (diff + '天后临产') : (diff === 0 ? '今日临产' : '已超期' + Math.abs(diff) + '天');
          changed.push(name + '=Đang Mang Thai(' + dueInfo.type + 'Ngày Dự Sinh' + dueStr + ',' + tail + ')');
        } else {
          syncPrivate(stat, name, { 'Kỳ Sinh Lý': PHASE_PREGNANT, 'Có Thai Không': '是' });
          changed.push(name + '=Kỳ Mang Thai，Ngày Dự Sinh等待确认');
        }
        return;
      }

      if (!rec) {
        rec = initRecord(name, sys, todayIdx, stat);
        changed.push(name + '=建档(' + rec['Giai Đoạn Hiện Tại'] + ',第' + rec['Ngày Thứ Mấy Chu Kỳ'] + '/' + rec['Độ Dài Chu Kỳ'] + '天)');
        return;
      }

      if (rec['Đang Mang Thai'] === true) {
        restartFromMenstruation(rec, todayIdx);
        changed.push(name + '=产后重启(Kỳ Kinh Nguyệt第1天)');
        syncPrivate(stat, name, { 'Kỳ Sinh Lý': rec['Giai Đoạn Hiện Tại'], 'Ngày Đặc Trưng Rụng Trứng': rec['Ngày Rụng Trứng'], 'Ngày Dự Sinh': '无', 'Loại Ngày Dự Sinh': '无' });
        return;
      }

      if (D > 0) {
        advanceRecord(rec, D, todayIdx);
        changed.push(name + '=+' + D + '天→' + rec['Giai Đoạn Hiện Tại'] + '(' + rec['Số Ngày Đến Giai Đoạn Tiếp Theo'] + '天后' + rec['Giai Đoạn Tiếp Theo'] + ')');
        syncPrivate(stat, name, { 'Kỳ Sinh Lý': rec['Giai Đoạn Hiện Tại'], 'Ngày Đặc Trưng Rụng Trứng': rec['Ngày Rụng Trứng'] });
      } else {
        refreshDerived(rec);
      }
    });

    sys['Sự Kiện Chờ Xử Lý'] = collectDueChecks(stat, todayIdx);
    sys['Chờ Thông Báo'] = collectPendingNotice(stat);

    const head = 'Ngày=' + raw + ' 基准=' + resolved.source + ' 推进=' + D + '天';
    return changed.length ? head + ' | ' + changed.join('；') : head;
  }

  function pickStat(variables) {
    if (!variables) return null;
    if (isPlainObject(variables.stat_data)) return variables.stat_data;
    if (isPlainObject(variables.variables?.stat_data)) return variables.variables.stat_data;
    if (isPlainObject(variables.data?.stat_data)) return variables.data.stat_data;
    if (isPlainObject(variables) && (isPlainObject(variables['Hệ Thống Chu Kỳ Sinh Lý']) || 'Ngày' in variables)) return variables;
    return null;
  }

  function pickStatContainer(source) {
    if (!isPlainObject(source)) return null;
    if (isPlainObject(source.stat_data)) return { holder: source, key: 'stat_data' };
    if (isPlainObject(source.variables?.stat_data)) return { holder: source.variables, key: 'stat_data' };
    if (isPlainObject(source.data?.stat_data)) return { holder: source.data, key: 'stat_data' };
    return null;
  }

  async function persistStat(stat, reason = 'cycle-update') {
    if (!isPlainObject(stat)) return false;
    const mvu = getMvu();
    let ok = false;
    try {
      if (mvu && typeof mvu.getMvuData === 'function' && typeof mvu.replaceMvuData === 'function') {
        const mvuData = JSON.parse(JSON.stringify(mvu.getMvuData({ type: 'message', message_id: 'latest' }) || {}));
        const container = pickStatContainer(mvuData);
        if (container) {
          container.holder[container.key] = JSON.parse(JSON.stringify(stat));
          await mvu.replaceMvuData(mvuData, { type: 'message', message_id: 'latest' });
          ok = true;
        }
      }
    } catch (e) {
      console.warn('[都市神壕 生理周期调度] Mvu persist failed:', e);
    }
    if (!ok) {
      const reader = getVariablesReader();
      const replacer = getVariablesReplacer();
      try {
        if (reader && replacer) {
          const vars = JSON.parse(JSON.stringify(reader({ type: 'message', message_id: 'latest' }) || {}));
          const container = pickStatContainer(vars);
          if (container) {
            container.holder[container.key] = JSON.parse(JSON.stringify(stat));
            replacer(vars, { type: 'message', message_id: 'latest' });
            ok = true;
          }
        }
      } catch (e) {
        console.warn('[都市神壕 生理周期调度] variables persist failed:', e);
      }
    }
    return ok;
  }

  function maintainCyclesSafe(stat) {
    try {
      return maintainCycles(stat);
    } catch (error) {
      console.warn('[都市神壕 生理周期调度] maintain failed:', error && error.message || error);
      return '';
    }
  }

  async function reconcileLatest(reason = 'boot') {
    let stat = null;
    const mvu = getMvu();
    try {
      if (mvu && typeof mvu.getMvuData === 'function') stat = pickStat(mvu.getMvuData({ type: 'message', message_id: 'latest' }));
    } catch (_) {}
    if (!stat) {
      const reader = getVariablesReader();
      if (reader) {
        try { stat = pickStat(reader({ type: 'message', message_id: 'latest' })); } catch (_) {}
      }
    }
    if (isPlainObject(stat)) {
      const log = maintainCyclesSafe(stat);
      if (log) {
        console.info('[都市神壕 生理周期调度] ' + reason + ' reconciliation applied:', log);
        await persistStat(stat, reason);
      }
    }
  }

  async function onVariables(variables) {
    try {
      const stat = pickStat(variables);
      const log = maintainCyclesSafe(stat);
      if (log) {
        console.info('[都市神壕 生理周期调度]', log);
        await persistStat(stat, 'VARIABLE_UPDATE_ENDED');
      }
    } catch (error) {
      console.warn('[都市神壕 生理周期调度] handler failed:', error && error.message || error);
    }
  }

  async function onGenerationEnded() {
    await reconcileLatest('GENERATION_ENDED');
  }

  const RETRY_LIMIT = 120;   // 250ms × 120 ≈ 30 秒
  const RETRY_GAP = 250;
  let bindDone = false;
  const bindStops = [];

  // 双挂点：变量更新完成（主） + 生成结束（兜底）。两者都只做幂等对账。
  function tryBind() {
    const mvu = getMvu();
    const on = getEventOn();
    if (!mvu || !mvu.events || typeof on !== 'function') return false;
    const stops = [];
    const add = (type, handler) => {
      if (!type) return false;
      try {
        const listener = on(type, handler);
        if (listener && typeof listener.stop === 'function') stops.push(() => { try { listener.stop(); } catch (_) {} });
        return true;
      } catch (error) {
        console.warn('[都市神壕 生理周期调度] bind failed:', type, error && error.message || error);
        return false;
      }
    };
    const okInit = add(mvu.events.VARIABLE_INITIALIZED, () => reconcileLatest('VARIABLE_INITIALIZED'));
    const okEnded = add(mvu.events.VARIABLE_UPDATE_ENDED, onVariables);
    const tav = getTavernEvents();
    const okGen = add(tav && tav.GENERATION_ENDED, onGenerationEnded);
    if (!okEnded && !okGen) {
      stops.splice(0).forEach(fn => fn());
      return false;
    }
    stops.forEach(fn => bindStops.push(fn));
    bindDone = true;
    const parts = [];
    if (okInit) parts.push('init');
    if (okEnded) parts.push('update_ended');
    if (okGen) parts.push('generation_ended');
    console.info('[都市神壕 生理周期调度] ready ' + CYCLE_VERSION + ' | bind=' + parts.join('+') + ' | mvu=' + mvuSource);
    return true;
  }

  function awaitMvu(attempt) {
    if (bindDone || tryBind()) return;
    if (attempt === 0) {
      try {
        if (typeof waitGlobalInitialized === 'function') Promise.resolve(waitGlobalInitialized('Mvu')).catch(() => {});
      } catch (_) {}
    }
    if (attempt >= RETRY_LIMIT) {
      console.error('[都市神壕 生理周期调度] 变量Sự Kiện挂载Thất Bại：本会话只在刷新页面时对账，游玩期间不推进。mvu=' + mvuSource + ' eventOn=' + (getEventOn() ? 'ok' : 'none'));
      return;
    }
    if (attempt > 0 && attempt % 20 === 0) {
      console.warn('[都市神壕 生理周期调度] 变量Sự Kiện仍在等待挂载 ' + attempt + '/' + RETRY_LIMIT);
    }
    HOST_WINDOW.setTimeout(() => awaitMvu(attempt + 1), RETRY_GAP);
  }

  async function waitOptionalGlobal(name, timeout = 1500) {
    if (name === 'Mvu' && getMvu()) return getMvu();
    if (window[name]) return window[name];
    try {
      if (typeof waitGlobalInitialized === 'function') {
        await Promise.race([
          waitGlobalInitialized(name),
          new Promise(resolve => setTimeout(resolve, timeout)),
        ]);
      }
    } catch (error) {
      console.warn('[都市神壕 生理周期调度] wait global failed:', name, error && error.message || error);
    }
    return name === 'Mvu' ? (getMvu() || null) : (window[name] || null);
  }

  async function boot() {
    await waitOptionalGlobal('Mvu');
    await reconcileLatest('boot');
    awaitMvu(0);
  }

  const runner = typeof errorCatched === 'function' ? errorCatched(boot) : boot;
  if (typeof $ === 'function') {
    $(runner);
  } else if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', runner);
  } else {
    runner();
  }
})();
