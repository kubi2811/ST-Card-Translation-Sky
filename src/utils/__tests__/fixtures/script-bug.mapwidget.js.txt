    // Thu phóng và di chuyển
    let mapScale = 1, mapX = 0, mapY = 0;
    function applyTransform() {
        mapSvg.style.transform = `translate(${mapX}px, ${mapY}px) scale(${mapScale})`;
    }
    function setScale(newScale, cx, cy) {
        const rect = mapViewport.getBoundingClientRect();
        if (cx == null) cx = rect.width / 2;
        if (cy == null) cy = rect.height / 2;
        const oldScale = mapScale;
        newScale = Math.max(0.5, Math.min(4, newScale));
        mapX = cx - (cx - mapX) * (newScale / oldScale);
        mapY = cy - (cy - mapY) * (newScale / oldScale);
        mapScale = newScale;
        applyTransform();
    }
    $('.zoom-in').onclick = () => setScale(mapScale * 1.3);
    $('.zoom-out').onclick = () => setScale(mapScale / 1.3);
    $('.zoom-reset').onclick = () => { mapScale = 1; mapX = 0; mapY = 0; applyTransform(); };

    mapViewport.addEventListener('wheel', (e) => {
        e.preventDefault();
        const rect = mapViewport.getBoundingClientRect();
        const cx = e.clientX - rect.left;
        const cy = e.clientY - rect.top;
        const delta = e.deltaY < 0 ? 1.15 : 1/1.15;
        setScale(mapScale * delta, cx, cy);
    }, { passive: false });

    let touchStartDist = null;
    mapViewport.addEventListener('touchstart', (e) => {
        if (e.touches.length === 1) {
            if (e.target.closest('.marker-group')) return;
            mapDragging = true;
            mapDragSX = e.touches[0].clientX; mapDragSY = e.touches[0].clientY;
            mapDragOX = mapX; mapDragOY = mapY;
        } else if (e.touches.length === 2) {
            mapDragging = false;
            const dx = e.touches[0].clientX - e.touches[1].clientX;
            const dy = e.touches[0].clientY - e.touches[1].clientY;
            touchStartDist = Math.sqrt(dx*dx + dy*dy);
        }
    }, { passive: true });
    mapViewport.addEventListener('touchmove', (e) => {
        if (e.touches.length === 1 && mapDragging) {
            mapX = mapDragOX + e.touches[0].clientX - mapDragSX;
            mapY = mapDragOY + e.touches[0].clientY - mapDragSY;
            applyTransform();
            e.preventDefault();
        } else if (e.touches.length === 2 && touchStartDist != null) {
            const dx = e.touches[0].clientX - e.touches[1].clientX;
            const dy = e.touches[0].clientY - e.touches[1].clientY;
            const dist = Math.sqrt(dx*dx + dy*dy);
            setScale(mapScale * (dist / touchStartDist));
            touchStartDist = dist;
            e.preventDefault();
        }
    }, { passive: false });
    mapViewport.addEventListener('touchend', () => {
        mapDragging = false; touchStartDist = null;

                        <button class="cfg-fetch-btn" id="cfg-fetch"><i class="fas fa-cloud-download-alt"></i>Lấy</button>

                    <label>Temperature</label>
                    <input type="number" id="cfg-temp" min="0" max="2" step="0.1"/>
                </div>
                <div class="cfg-switch">
                    <label style="font-size:13px;color:var(--text-main);">Phản hồi dạng luồng (hiệu ứng máy đánh chữ)</label>
                    <label class="switch">
                        <input type="checkbox" id="cfg-stream"/>
                        <span class="slider"></span>
                    </label>
                </div>
                <div class="cfg-btn-row">
                    <button class="cfg-cancel" id="cfg-cancel">Hủy</button>
                    <button class="cfg-save" id="cfg-save">Lưu</button>
                </div>
            </div>
        `;
        modalMask.classList.add('show');

        const providerSel = modal.querySelector('#cfg-provider');
        const urlIn = modal.querySelector('#cfg-url');
        const keyIn = modal.querySelector('#cfg-key');
        const modelIn = modal.querySelector('#cfg-model');
        const modelSelect = modal.querySelector('#cfg-model-select');
        const tempIn = modal.querySelector('#cfg-temp');
        const streamIn = modal.querySelector('#cfg-stream');
        const fetchBtn = modal.querySelector('#cfg-fetch');

        providerSel.value = apiConfig.provider;
        urlIn.value = apiConfig.baseUrl;
        keyIn.value = apiConfig.apiKey;
        modelIn.value = apiConfig.model;
        tempIn.value = apiConfig.temperature;
        streamIn.checked = apiConfig.stream;

        providerSel.onchange = () => {
            const preset = providerPresets[providerSel.value];
            if (preset) {
                if (preset.baseUrl) urlIn.value = preset.baseUrl;
                if (preset.model) modelIn.value = preset.model;
            }
            modelSelect.style.display = 'none';
            modelIn.style.display = '';
        };

        fetchBtn.onclick = async () => {
            const url = urlIn.value.trim().replace(/\/$/, '');
            const key = keyIn.value.trim();
            if (!url || !key) {
                alert('Vui lòng điền Base URL và API Key trước');
                return;
            }
            fetchBtn.disabled = true;
            fetchBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i>Đang lấy';
            try {
                const res = await fetch(`${url}/models`, {
                    headers: { 'Authorization': `Bearer ${key}` }
                });
                if (!res.ok) throw new Error('HTTP ' + res.status);
                const data = await res.json();
                const models = (data.data || data.models || []).map(m => m.id || m.name || m).filter(Boolean);
                if (models.length === 0) throw new Error('Không trả về mô hình nào');
                modelSelect.innerHTML = models.map(m => `<option value="${m}">${m}</option>`).join('');
                if (models.includes(modelIn.value)) modelSelect.value = modelIn.value;
                modelSelect.style.display = '';
                modelIn.style.display = 'none';
                modelSelect.onchange = () => { modelIn.value = modelSelect.value; };
                modelIn.value = modelSelect.value;
                fetchBtn.innerHTML = '<i class="fas fa-check"></i>Đã lấy';
                setTimeout(() => {
                    fetchBtn.innerHTML = '<i class="fas fa-cloud-download-alt"></i>Lấy';
                    fetchBtn.disabled = false;
                }, 1500);
            } catch (e) {
                console.error(e);
                alert('Lấy mô hình thất bại: ' + e.message);
                fetchBtn.innerHTML = '<i class="fas fa-cloud-download-alt"></i>Lấy';
                fetchBtn.disabled = false;
            }
        };modal.querySelector('#cfg-cancel').onclick = () => modalMask.classList.remove('show');
        modal.querySelector('#cfg-save').onclick = () => {
            apiConfig = {
                provider: providerSel.value,
                baseUrl: urlIn.value.trim().replace(/\/$/, ''),
                apiKey: keyIn.value.trim(),
                model: modelIn.value.trim(),
                temperature: parseFloat(tempIn.value) || 0.7,
                stream: streamIn.checked
            };
            try { localStorage.setItem(CONFIG.API_KEY, JSON.stringify(apiConfig)); } catch (e) {}
            modalMask.classList.remove('show');
            if (typeof triggerSlash === 'function') triggerSlash('/echo severity=success Cấu hình API đã được lưu');
        };
    }
    cfgBtn.onclick = showConfigModal;
    panel.querySelector('.cfg-link').onclick = showConfigModal;
