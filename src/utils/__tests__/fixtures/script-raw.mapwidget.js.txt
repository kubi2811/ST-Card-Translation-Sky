    let mapScale = 1, mapX = 0, mapY = 0;
    function applyTransform() {
        mapSvg.style.transform = `translate(${mapX}px, ${mapY}px) scale(${mapScale})`;
    }
    function setScale(newScale, cx, cy) {
        const rect = mapViewport.getBoundingClientRect();
        if (cx == null) cx = rect.width / 2;
        if (cy == null) cy = rect.height / 2;
        const oldScale = mapScale;
        newScale = Math.max(0.5, Math.min(4, newScale));
        mapX = cx - (cx - mapX) * (newScale / oldScale);
        mapY = cy - (cy - mapY) * (newScale / oldScale);
        mapScale = newScale;
        applyTransform();
    }
    $('.zoom-in').onclick = () => setScale(mapScale * 1.3);
    $('.zoom-out').onclick = () => setScale(mapScale / 1.3);

        const cx = e.clientX - rect.left;
        const cy = e.clientY - rect.top;

    const REGEX_MAP = {
        npc: /<npc>([\s\S]*?)<\/npc>/i,
        place: /<place>([\s\S]*?)<\/place>/i,
        thing: /<thing>([\s\S]*?)<\/thing>/i,
        other: /<other>([\s\S]*?)<\/other>/i
    };
    const OPEN_TAG = { npc: '<npc>', place: '<place>', thing: '<thing>', other: '<other>' };
    const CLOSE_TAG = { npc: '</npc>', place: '</place>', thing: '</thing>', other: '</other>' };

    function extractTagContent(cat, raw) {
        if (!raw) return null;
        let cleaned = raw.replace(/<think>[\s\S]*?<\/think>/gi, '');
        cleaned = cleaned.replace(/^[\s\S]*?<\/think>/i, '');
        const match = cleaned.match(REGEX_MAP[cat]);
        if (match && match[1]) {
            const content = match[1].trim();
            if (/^[（(].*[）)][（(].*[）)]/.test(content) && content.length < 50) return null;
            return content;
        }
        return null;
    }

    // 流式过程中的部分内容提取（用于打字机）
    function extractPartial(cat, buffer) {
        if (!buffer) return '';
        let cleaned = buffer.replace(/<think>[\s\S]*?<\/think>/gi, '');
        cleaned = cleaned.replace(/^[\s\S]*?<\/think>/i, '');
        const openIdx = cleaned.indexOf(OPEN_TAG[cat]);
        if (openIdx < 0) return '';
        const start = openIdx + OPEN_TAG[cat].length;
        const closeIdx = cleaned.indexOf(CLOSE_TAG[cat], start);
        if (closeIdx >= 0) {
            return cleaned.substring(start, closeIdx).trim();
        }
        return cleaned.substring(start).trim();
    }

    async function callApiStream(prompt, onDelta, signal) {
        const url = `${apiConfig.baseUrl}/chat/completions`;
        const res = await fetch(url, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${apiConfig.apiKey}`
            },
            body: JSON.stringify({
                model: apiConfig.model,
                messages: [{ role: 'user', content: prompt }],
                temperature: apiConfig.temperature,
                stream: true
            }),
            signal
        });
        if (!res.ok) {
            const errText = await res.text().catch(() => '');
            throw new Error(`HTTP ${res.status}: ${errText.slice(0, 200)}`);
        }
        const reader = res.body.getReader();
        const decoder = new TextDecoder('utf-8');
        let buffer = '';
        let full = '';
        while (true) {
            const { value, done } = await reader.read();
            if (done) break;
            buffer += decoder.decode(value, { stream: true });
            const lines = buffer.split('\n');
            buffer = lines.pop() || '';
            for (const line of lines) {
                const trimmed = line.trim();
                if (!trimmed || !trimmed.startsWith('data:')) continue;
                const data = trimmed.slice(5).trim();
                if (data === '[DONE]') continue;
                try {
                    const parsed = JSON.parse(data);
                    const delta = parsed.choices?.[0]?.delta?.content || '';
                    if (delta) {
                        full += delta;
                        onDelta(full);
                    }
                } catch (e) {}
            }
        }
        return full;
    }

            }),
            signal
        });
        if (!res.ok) {
            const errText = await res.text().catch(() => '');
            throw new Error(`HTTP ${res.status}: ${errText.slice(0, 200)}`);
        }
        const data = await res.json();
        return data.choices?.[0]?.message?.content || '';
    }

    async function handleSearch() {
        const q = searchInput.value.trim();
        if (!q || isGenerating) return;

        if (!apiConfig.apiKey || !apiConfig.baseUrl || !apiConfig.model) {
            if (typeof triggerSlash === 'function') triggerSlash('/echo severity=warning 请先配置 API');
            showConfigModal();
            return;
        }

        isGenerating = true;
        searchBtn.disabled = true;
        searchInput.disabled = true;

        const hint = resultsArea.querySelector('.empty-hint');
        if (hint) hint.remove();

        const cat = currentCategory;
        const prompt = PROMPTS[cat](q);
        const card = renderCardShell(cat);
        const cardBody = card.querySelector('.card-body');
        resultsArea.insertBefore(card, resultsArea.firstChild);

        // 显示初始加载点
        cardBody.innerHTML = `<span style="color:var(--text-muted);">正在检索……</span>`;

        currentAbortController = new AbortController();
        let raw = '';

        try {
            if (apiConfig.stream) {
                raw = await callApiStream(prompt, (accumulated) => {
                    const partial = extractPartial(cat, accumulated);
                    if (partial) {
                        cardBody.textContent = partial;
                    }
                }, currentAbortController.signal);
            } else {
                cardBody.classList.remove('streaming');
                cardBody.innerHTML = `<div style="text-align:center;color:var(--text-muted);"><div class="typing-dots"><span></span><span></span><span></span></div></div>`;
                raw = await callApiNonStream(prompt, currentAbortController.signal);
            }

            const content = extractTagContent(cat, raw);
            cardBody.classList.remove('streaming');
            if (content) {
                cardBody.textContent = content;
            } else {
                cardBody.textContent = raw ? '（未能解析出结构化结果，原文如下）\n\n' + raw.trim() : '（未获取到有效回复）';
            }
        } catch (e) {
            cardBody.classList.remove('streaming');
            if (e.name === 'AbortError') {
                cardBody.textContent = '（已取消）';
            } else {
                cardBody.textContent = '请求失败：' + e.message;
                console.error('[Search API]', e);
            }
        } finally {
            currentAbortController = null;
            searchInput.value = '';
            isGenerating = false;
            searchBtn.disabled = false;
            searchInput.disabled = false;
            searchInput.focus();
        }
    }
    searchBtn.onclick = handleSearch;
    searchInput.onkeypress = (e) => {
        if (e.key === 'Enter') { e.preventDefault(); handleSearch(); }
    };

    function clamp(el, isBall) {
        const w = el.offsetWidth, h = el.offsetHeight;
        let l = parseFloat(el.style.left) || 0;
        let t = parseFloat(el.style.top) || 0;
        const maxL = tWin.innerWidth - w - CONFIG.SAFE_MARGIN;
        const maxT = isBall ? tWin.innerHeight - h - CONFIG.SAFE_MARGIN : tWin.innerHeight - 50;
        el.style.left = Math.max(CONFIG.SAFE_MARGIN, Math.min(l, maxL)) + 'px';
        el.style.top = Math.max(CONFIG.SAFE_MARGIN, Math.min(t, maxT)) + 'px';
    }

    tWin.addEventListener('resize', () => {
        if (!isVisible) return;
        if (currentState === STATES.PANEL_OPEN) clamp(panel, false);
        if (currentState === STATES.IDLE) clamp(ball, true);
    });

    function makeDrag(el, handle, isBall) {
        let sx, sy, il, it, dragging = false, raf, downTime;
        const onDown = (e) => {
            if (isBall && currentState !== STATES.IDLE) return;
            if (!isBall && currentState !== STATES.PANEL_OPEN) return;
            const ev = e.touches ? e.touches[0] : e;
            sx = ev.clientX; sy = ev.clientY;
            const rect = el.getBoundingClientRect();
            il = rect.left; it = rect.top;
            dragging = false; downTime = Date.now();
            tDoc.addEventListener('mousemove', onMove, { passive: false });
            tDoc.addEventListener('touchmove', onMove, { passive: false });
            tDoc.addEventListener('mouseup', onUp);
            tDoc.addEventListener('touchend', onUp);
        };
        const onMove = (e) => {
            const ev = e.touches ? e.touches[0] : e;
            const dx = ev.clientX - sx, dy = ev.clientY - sy;
            if (!dragging) {
                if (Math.sqrt(dx*dx + dy*dy) > CONFIG.CLICK_DIST) {
                    dragging = true;
                    el.classList.add('dragging');
                    currentState = isBall ? STATES.BALL_DRAG : STATES.PANEL_DRAG;
                } else return;
            }
            e.preventDefault();
            if (raf) cancelAnimationFrame(raf);
            raf = requestAnimationFrame(() => {
                let nl = il + dx, nt = it + dy;
                const w = el.offsetWidth, h = el.offsetHeight;
                const maxL = tWin.innerWidth - w - CONFIG.SAFE_MARGIN;
                const maxT = isBall ? tWin.innerHeight - h - CONFIG.SAFE_MARGIN : tWin.innerHeight - 50;
                el.style.left = Math.max(CONFIG.SAFE_MARGIN, Math.min(nl, maxL)) + 'px';
                el.style.top = Math.max(CONFIG.SAFE_MARGIN, Math.min(nt, maxT)) + 'px';
            });
        };
        const onUp = () => {
            tDoc.removeEventListener('mousemove', onMove);
            tDoc.removeEventListener('touchmove', onMove);
            tDoc.removeEventListener('mouseup', onUp);
            tDoc.removeEventListener('touchend', onUp);
            if (raf) cancelAnimationFrame(raf);
            const elapsed = Date.now() - downTime;
            if (!dragging && elapsed < 500) {
                if (isBall) openPanel();
            } else if (dragging) {
                el.classList.remove('dragging');
                if (isBall) {
                    snapEdge();
                    try {
                        localStorage.setItem(CONFIG.STORAGE_KEY, JSON.stringify({
                            left: parseFloat(el.style.left), top: parseFloat(el.style.top)
                        }));
                    } catch (e) {}
                } else {
                    currentState = STATES.PANEL_OPEN;
                    clamp(el, false);
                }
            }
            dragging = false;
        };
        handle.addEventListener('mousedown', onDown);
        handle.addEventListener('touchstart', onDown, { passive: true });
    }

    function snapEdge() {
        const rect = ball.getBoundingClientRect();
        const targetL = (rect.left + rect.width / 2) < tWin.innerWidth / 2
            ? CONFIG.SAFE_MARGIN
            : tWin.innerWidth - rect.width - CONFIG.SAFE_MARGIN;
        ball.style.left = targetL + 'px';
        setTimeout(() => { currentState = STATES.IDLE; }, 350);
    }

    function openPanel() {
        currentState = STATES.PANEL_OPEN;
        ball.classList.add('hidden');
        panel.classList.add('show');
        const br = ball.getBoundingClientRect();
        const pw = panel.offsetWidth, ph = panel.offsetHeight;
        let pl = br.left - pw / 2 + br.width / 2;
        let pt = br.top - ph - 10;
        if (pt < CONFIG.SAFE_MARGIN) pt = br.bottom + 10;
        pl = Math.max(CONFIG.SAFE_MARGIN, Math.min(pl, tWin.innerWidth - pw - CONFIG.SAFE_MARGIN));
        pt = Math.max(CONFIG.SAFE_MARGIN, Math.min(pt, tWin.innerHeight - ph - CONFIG.SAFE_MARGIN));
        panel.style.left = pl + 'px';
        panel.style.top = pt + 'px';
        startParticles();
    }

    function closePanel() {
        currentState = STATES.IDLE;
        panel.classList.remove('show');
        ball.classList.remove('hidden');
        modalMask.classList.remove('show');
        stopParticles();
        if (isVisible) clamp(ball, true);
    }

    closeBtn.onclick = closePanel;
    closeBtn.addEventListener('touchend', (e) => { e.preventDefault(); closePanel(); });

    (function setupResize() {
        let sx, sy, sw, sh;
        const onDown = (e) => {
            const ev = e.touches ? e.touches[0] : e;
            sx = ev.clientX; sy = ev.clientY;
            sw = panel.offsetWidth; sh = panel.offsetHeight;
            panel.style.transition = 'none';
            tDoc.addEventListener('mousemove', onMove, { passive: false });
            tDoc.addEventListener('touchmove', onMove, { passive: false });
            tDoc.addEventListener('mouseup', onUp);
            tDoc.addEventListener('touchend', onUp);
            e.stopPropagation(); e.preventDefault();
        };
        const onMove = (e) => {
            const ev = e.touches ? e.touches[0] : e;
            requestAnimationFrame(() => {
                const cl = parseFloat(panel.style.left) || 0;
                const ct = parseFloat(panel.style.top) || 0;
                let nw = Math.max(340, sw + ev.clientX - sx);
                let nh = Math.max(480, sh + ev.clientY - sy);
                nw = Math.min(nw, tWin.innerWidth - cl - CONFIG.SAFE_MARGIN);
                nh = Math.min(nh, tWin.innerHeight - ct - CONFIG.SAFE_MARGIN);
                panel.style.width = nw + 'px';
                panel.style.height = nh + 'px';
            });
        };
        const onUp = () => {
            tDoc.removeEventListener('mousemove', onMove);
            tDoc.removeEventListener('touchmove', onMove);
            tDoc.removeEventListener('mouseup', onUp);
            tDoc.removeEventListener('touchend', onUp);
            panel.style.transition = '';
            clamp(panel, false);
        };
        resizeHandle.addEventListener('mousedown', onDown);
        resizeHandle.addEventListener('touchstart', onDown, { passive: false });
    })();

    makeDrag(ball, ball, true);
    makeDrag(panel, header, false);

    if (typeof appendInexistentScriptButtons === 'function') {
        appendInexistentScriptButtons([{ name: '显隐悬浮球', visible: true }]);
    }
    if (typeof eventOn === 'function' && typeof getButtonEvent === 'function') {
        eventOn(getButtonEvent('显隐悬浮球'), () => {
            isVisible = !isVisible;
            if (isVisible) {
                ball.classList.remove('force-hide');
                panel.classList.remove('force-hide');
                if (typeof triggerSlash === 'function') triggerSlash('/echo severity=info 悬浮球已显示');
            } else {
                ball.classList.add('force-hide');
                panel.classList.add('force-hide');
                if (typeof triggerSlash === 'function') triggerSlash('/echo severity=info 悬浮球已隐藏');
            }
        });
    }
})();
